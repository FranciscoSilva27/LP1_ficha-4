import java.util.Scanner;
import java.lang.Math;
import java.util.Arrays; // Para usar Arrays.sort

public class exercicio4g2_2 {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        System.out.println("--- Análise Geométrica Completa ---");
        
        // 1. Ler os 3 lados
        System.out.print("Introduza o comprimento do Lado A (inteiro): ");
        int a = lerInteiro(scanner);
        
        System.out.print("Introduza o comprimento do Lado B (inteiro): ");
        int b = lerInteiro(scanner);
        
        System.out.print("Introduza o comprimento do Lado C (inteiro): ");
        int c = lerInteiro(scanner);

        // 2. Aplicar a Regra da Desigualdade Triangular
        boolean podeFormarTriangulo = (a + b > c) && (a + c > b) && (b + c > a);

        System.out.println("\n--- Resultado da Análise ---");
        System.out.printf("Lados lidos: A=%d, B=%d, C=%d\n", a, b, c);
        
        if (podeFormarTriangulo) {
            System.out.println("✅ Os valores PODEM formar um triângulo.");
            
            // 3. Classificação (mantida)
            if (a == b && b == c) {
                System.out.println("Tipo: EQUILÁTERO");
            } else if (a == b || a == c || b == c) {
                System.out.println("Tipo: ISÓSCELES");
            } else {
                System.out.println("Tipo: ESCALENO");
            }
            
            // 4. Cálculo da Área (Heron) (mantido)
            double s = (double) (a + b + c) / 2;
            double areaTriangulo = Math.sqrt(s * (s - a) * (s - b) * (s - c));
            
            System.out.println("\n--- Cálculo da Área (Heron) ---");
            System.out.printf("Área do Triângulo: %.2f unidades quadradas\n", areaTriangulo);
            
            // NOVO: ANÁLISE DO RETÂNGULO ASSOCIADO
            
            // 5. Determinar a Base (Usamos o lado mais longo como Base b_max)
            int[] lados = {a, b, c};
            Arrays.sort(lados); // Ordena para encontrar o maior lado
            double baseRetangulo = lados[2]; // O maior lado está no índice 2
            
            // 6. Calcular a Altura relativa a essa base
            // h = (2 * Area) / Base
            double alturaRetangulo = (2 * areaTriangulo) / baseRetangulo;
            
            // 7. Calcular a Área do Retângulo (deve ser o dobro da área do triângulo)
            double areaRetangulo = baseRetangulo * alturaRetangulo;
            
            System.out.println("\n--- Retângulo Associado (Base = lado mais longo) ---");
            System.out.printf("Base do Retângulo (Lado Máximo do Triângulo): %.0f\n", baseRetangulo);
            System.out.printf("Altura do Retângulo (Altura relativa à Base): %.2f\n", alturaRetangulo);
            System.out.printf("Área do Retângulo (Base x Altura): %.2f unidades quadradas\n", areaRetangulo);
            
        } else {
            System.out.println("❌ Os valores NÃO podem formar um triângulo.");
        }

        scanner.close();
    }
    
    // Método auxiliar (Mantido)
    private static int lerInteiro(Scanner scanner) {
        int valor = -1;
        while (valor <= 0) {
            if (scanner.hasNextInt()) {
                valor = scanner.nextInt();
                if (valor <= 0) {
                    System.out.print("O lado deve ser um valor inteiro positivo. Introduza novamente: ");
                }
            } else {
                System.out.print("Entrada inválida. Introduza um número inteiro positivo: ");
                scanner.next(); 
            }
        }
        return valor;
    }
}
