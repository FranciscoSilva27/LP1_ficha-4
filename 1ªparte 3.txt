import java.util.Scanner;

public class Exercicio4g1 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n;                  // quantidade de números a ler
        int numero;             // número lido

        int qtPositivos = 0;    // contador de números positivos
        int soma = 0;           // soma de todos os números

        int qtNegativos = 0;    // contador de números negativos
        int somaNegativos = 0;  // soma dos números negativos

        // Validação do N (tem de ser positivo)
        do {
            System.out.print("Quantos numeros pretende inserir (N)? ");
            n = sc.nextInt();

            if (n <= 0) {
                System.out.println("Erro: N deve ser um inteiro positivo. Tente novamente.");
            }
        } while (n <= 0);

        // Leitura dos N números
        for (int i = 1; i <= n; i++) {
            System.out.print("Introduza o " + i + "º numero inteiro: ");
            numero = sc.nextInt();

            // soma total
            soma += numero;

            // conta positivos
            if (numero > 0) {
                qtPositivos++;
            }

            // soma e conta negativos
            if (numero < 0) {
                somaNegativos += numero;
                qtNegativos++;
            }
        }

        // Cálculo da percentagem de positivos
        double percentagemPositivos = (qtPositivos * 100.0) / n;

        // Cálculo da média de todos os números
        double mediaGeral = soma / (double) n;

        // Cálculo da média dos negativos (se houver)
        double mediaNegativos = 0;
        if (qtNegativos > 0) {
            mediaNegativos = somaNegativos / (double) qtNegativos;
        }

        System.out.printf("Percentagem de numeros positivos: %.2f%%%n", percentagemPositivos);
        System.out.println("Soma de todos os numeros lidos: " + soma);
        System.out.printf("Media geral dos numeros lidos: %.2f%n", mediaGeral);

        System.out.println("Soma dos numeros negativos: " + somaNegativos);
        if (qtNegativos > 0) {
            System.out.printf("Media dos numeros negativos: %.2f%n", mediaNegativos);
        } else {
            System.out.println("Nao foram lidos numeros negativos.");
        }

        sc.close();
    }
}
