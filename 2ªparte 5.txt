import java.util.Scanner;
import java.lang.Math;
import java.util.Arrays; 

public class exercicio4g2_2 {

    // Método auxiliar para verificar se um número é primo
    private static boolean ehPrimo(int n) {
        // 1. Caso base: 1 e números menores ou iguais a 1 não são primos
        if (n <= 1) {
            return false;
        }
        // 2. Verifica divisibilidade de 2 até a raiz quadrada de n
        // Se n for divisível por algum número até Math.sqrt(n), ele não é primo.
        for (int i = 2; i <= Math.sqrt(n); i++) {
            if (n % i == 0) {
                return false;
            }
        }
        return true;
    }
    
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        System.out.println("--- Análise Geométrica e Primalidade ---");
        
        // 1. Ler os 3 lados
        System.out.print("Introduza o comprimento do Lado A (inteiro): ");
        int a = lerInteiro(scanner);
        
        System.out.print("Introduza o comprimento do Lado B (inteiro): ");
        int b = lerInteiro(scanner);
        
        System.out.print("Introduza o comprimento do Lado C (inteiro): ");
        int c = lerInteiro(scanner);

        // [...] Código de Verificação e Classificação do Triângulo omitido para brevidade
        // (O seu código deve manter esta parte)
        boolean podeFormarTriangulo = (a + b > c) && (a + c > b) && (b + c > a);
        double areaTriangulo = 0.0;
        
        if (podeFormarTriangulo) {
            double s = (double) (a + b + c) / 2;
            areaTriangulo = Math.sqrt(s * (s - a) * (s - b) * (s - c));
            
            // ... (Impressão da Área e Retângulo associado)
            int[] lados = {a, b, c};
            Arrays.sort(lados);
            double baseRetangulo = lados[2];
            double alturaRetangulo = (2 * areaTriangulo) / baseRetangulo;
            double areaRetangulo = baseRetangulo * alturaRetangulo;
            
            System.out.println("\n--- Análise do Triângulo ---");
            System.out.printf("Área do Triângulo: %.2f\n", areaTriangulo);
            System.out.printf("Base Retângulo Associado: %.0f, Altura: %.2f\n", baseRetangulo, alturaRetangulo);
        } else {
            System.out.println("❌ Os valores NÃO podem formar um triângulo.");
        }
        
        // NOVO: VERIFICAÇÃO DE PRIMALIDADE
        System.out.println("\n--- Verificação de Primalidade dos Lados ---");
        
        System.out.println("Lado A (" + a + "): " + (ehPrimo(a) ? "✅ É PRIMO" : "❌ NÃO É PRIMO"));
        System.out.println("Lado B (" + b + "): " + (ehPrimo(b) ? "✅ É PRIMO" : "❌ NÃO É PRIMO"));
        System.out.println("Lado C (" + c + "): " + (ehPrimo(c) ? "✅ É PRIMO" : "❌ NÃO É PRIMO"));

        scanner.close();
    }
    
    // Método auxiliar para ler e validar que a entrada é um inteiro positivo (Mantido)
    private static int lerInteiro(Scanner scanner) {
        int valor = -1;
        while (valor <= 0) {
            if (scanner.hasNextInt()) {
                valor = scanner.nextInt();
                if (valor <= 0) {
                    System.out.print("O lado deve ser um valor inteiro positivo (> 0). Introduza novamente: ");
                }
            } else {
                System.out.print("Entrada inválida. Introduza um número inteiro positivo: ");
                scanner.next(); 
            }
        }
        return valor;
    }
}
